# =============================================================================
# SENA Sistema de Gestión de Maquinaria — Dependencias del Proyecto
# =============================================================================

# -----------------------------------------------------------------------------
# Framework principal
# -----------------------------------------------------------------------------
Django                  # Framework web principal — manejo de vistas, modelos, ORM, admin y autenticación

# -----------------------------------------------------------------------------
# Extensiones de Django
# -----------------------------------------------------------------------------
djangorestframework     # Construcción de la API REST (serializers, ViewSets, autenticación por token)
django-cors-headers     # Habilita CORS para permitir peticiones desde otros dominios (frontend externo o apps móviles)
django-environ          # Carga variables de entorno desde archivo .env (credenciales, claves secretas)

# -----------------------------------------------------------------------------
# Base de datos
# -----------------------------------------------------------------------------
mysqlclient             # Conector nativo entre Django y MySQL

# -----------------------------------------------------------------------------
# Manejo de archivos e imágenes
# -----------------------------------------------------------------------------
Pillow                  # Procesamiento de imágenes: fotos de máquinas, perfiles de usuario y piezas de inventario

# -----------------------------------------------------------------------------
# Generación y exportación de reportes
# -----------------------------------------------------------------------------
openpyxl                # Exportación e importación de datos en formato Excel (.xlsx)
reportlab               # Generación de reportes y fichas técnicas en formato PDF

# -----------------------------------------------------------------------------
# Extracción de contenido de documentos
# -----------------------------------------------------------------------------
PyMuPDF                 # Lectura de archivos PDF: extracción de texto, índice (TOC) y metadata
python-docx             # Lectura de archivos Word (.docx): extracción de headings y contenido para indexación

# -----------------------------------------------------------------------------
# Reconocimiento facial
# -----------------------------------------------------------------------------
opencv-python           # Captura y preprocesamiento de imágenes desde la cámara web
numpy                   # Operaciones matemáticas sobre vectores de embeddings faciales (128 dimensiones)
deepface                # Extracción del embedding facial y comparación de similitud entre rostros
tf-keras                # Backend de TensorFlow/Keras requerido por DeepFace para los modelos de reconocimiento

# -----------------------------------------------------------------------------
# Tareas asíncronas y caché
# -----------------------------------------------------------------------------
celery                  # Ejecución de tareas en segundo plano: generación de reportes, indexación de documentos
redis                   # Broker de mensajes para Celery y almacenamiento en caché de sesiones y consultas

# -----------------------------------------------------------------------------
# Utilidades generales
# -----------------------------------------------------------------------------
python-dateutil         # Manejo avanzado de fechas, zonas horarias y parsing de strings de fecha
requests                # Realización de peticiones HTTP externas desde el servidor
